{{- /* Site override of the theme's image hook (themes/blowfish/layouts/_default/_markup/render-image.html). Differences: - Photos are turned upright by their EXIF orientation, so portrait phone photos no longer show up sideways. - The article and the zoom view only use processed copies (up to 1280 px in the article, 2400 px when zoomed), which carry no camera metadata. - `sizes` matches this site's 42rem reading column. - Portrait images stay within the viewport height (custom.css, --img-ratio). - A relative image path that matches no file prints a warning. Images that still contain GPS data stop the build: partials/blog/image-guard.html. */ -}} {{- $url := urls.Parse .Destination -}} {{- $alt := .PlainText -}} {{- $res := "" -}} {{- if not $url.Scheme -}} {{- $res = or (.Page.Resources.Get $url.Path) (resources.Get (strings.TrimPrefix "/" $url.Path)) -}} {{- if and (not $res) (not (strings.HasPrefix $url.Path "/")) -}} {{- warnf "Image not found in %q: %q (put the file in the post's folder)" .Page.Path .Destination -}} {{- end -}} {{- end -}} {{- $isRaster := false -}} {{- with $res -}} {{- $isRaster = and (eq .MediaType.MainType "image") (not (in (slice "svg" "svg+xml" "gif") .MediaType.SubType)) -}} {{- end -}}
