{{- /* Previous / next navigation at the end of a post. Posts in a series follow the series (chapter) order. Other posts fall back to their category, ordered by date (older = previous). Params (dict): page, series (from blog/series.html) */ -}} {{- $page := .page -}} {{- $series := .series -}} {{- $show := $page.Params.showPagination | default (site.Params.article.showPagination | default true) -}} {{- if $show -}} {{- $prev := false -}} {{- $next := false -}} {{- $prevChapter := "" -}} {{- $nextChapter := "" -}} {{- if $series.current -}} {{- with $series.prev }}{{ $prev = .page }}{{ $prevChapter = .chapter }}{{ end -}} {{- with $series.next }}{{ $next = .page }}{{ $nextChapter = .chapter }}{{ end -}} {{- else if $page.CurrentSection -}} {{- if not $page.CurrentSection.IsHome -}} {{- $pages := $page.CurrentSection.RegularPages -}} {{- $siblings := (union (where $pages "Params.chapter" "==" nil) (where $pages "Params.chapter" "==" "")).ByDate -}} {{- $prev = $siblings.Prev $page -}} {{- $next = $siblings.Next $page -}} {{- end -}} {{- end -}} {{- if or $prev $next -}}

{{- end -}} {{- with $series.section }}
