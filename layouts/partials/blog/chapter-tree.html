      {{- end -}} {{- else -}} {{- if gt $depth $d -}} {{- range seq (sub $depth $d) -}}
    {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $depth = $d -}} {{- $p := $it.page -}} {{- $isCurrent := eq $p.RelPermalink $currentURL -}} {{- $isAncestor := and $currentChapter (hasPrefix $currentChapter (printf "%s." $it.chapter)) -}}
